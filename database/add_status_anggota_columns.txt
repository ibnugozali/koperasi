-- SQL untuk menambahkan kolom status_anggota dan data_keluar ke tabel anggota
-- Jalankan SQL ini jika mendapat error saat mengajukan keluar dari koperasi

-- Tambahkan kolom status_anggota jika belum ada
DO $$ 
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM information_schema.columns 
        WHERE table_name = 'anggota' AND column_name = 'status_anggota'
    ) THEN
        ALTER TABLE anggota ADD COLUMN status_anggota VARCHAR(50) DEFAULT NULL;
        RAISE NOTICE 'Kolom status_anggota berhasil ditambahkan';
    ELSE
        RAISE NOTICE 'Kolom status_anggota sudah ada';
    END IF;
END $$;

-- Tambahkan kolom data_keluar jika belum ada
DO $$ 
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM information_schema.columns 
        WHERE table_name = 'anggota' AND column_name = 'data_keluar'
    ) THEN
        ALTER TABLE anggota ADD COLUMN data_keluar JSONB DEFAULT NULL;
        RAISE NOTICE 'Kolom data_keluar berhasil ditambahkan';
    ELSE
        RAISE NOTICE 'Kolom data_keluar sudah ada';
    END IF;
END $$;

-- Tambahkan comment untuk dokumentasi
COMMENT ON COLUMN anggota.status_anggota IS 'Status khusus anggota: pending_keluar, keluar, dll. NULL untuk anggota aktif biasa';
COMMENT ON COLUMN anggota.data_keluar IS 'Data JSON untuk pengajuan keluar: simpanan_wajib, simpanan_lainnya, biaya_admin, alasan, tanggal_pengajuan';
